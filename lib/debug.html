5template <class T>
6concept Range = requires(const T &x) { begin(x); };
7template <class T>
8concept Tuple = requires { tuple_size<remove_cvref_t<T>>::value; };
9template <class T>
10concept Text = requires(const T &x) { string_view(x); };
11template <class T>
12void Dprint(const T &x) {
13if constexpr (Text<T>) {
14cerr << x;
15} else if constexpr (Range<T>) {
16cerr << '{';
17bool sep = 0;
18for (const auto &e : x) {
19if (sep) cerr << ",";
20Dprint(e);
21sep = 1;
22}
23cerr << '}';
24} else if constexpr (Tuple<T>) {
25cerr << '{';
26bool sep = 0;
27auto one = [&](const auto &e) {
28if (sep) cerr << ",";
29Dprint(e);
30sep = 1;
31};
32apply([&](const auto &...e) { (one(e), ...); }, x);
33cerr << '}';
34} else {
35cerr << x;
36}
37}
38template <class... T>
39void print(int line, const T &...args) {
40cerr << '[' << line << "] ";
41size_t i = 0;
42auto one = [&](const auto &x) {
43if (i) cerr << ',';
44const bool color = i++ & 1;
45cerr << (color ? "\033[96m" : "");
46Dprint(x);
47cerr << (color ? "\033[0m" : "");
48};
49(one(args), ...);
50cerr << '\n';
51}
