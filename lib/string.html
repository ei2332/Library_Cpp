文字列アルゴリズム（Manacher・Z Algorithm）

各位置を中心とする最長の奇数長回文の半径（中心を含む）
s = ababa -> {1, 2, 3, 2, 1}
偶数長回文は文字間にダミー文字を挿入して求める
計算量 O(N)
1vector<int> manacher(const string& s) {
2 int n = s.size(), i = 0, j = 1, k;
3 vector<int> res(n);
4 while (i < n) {
5 while (i >= j && i + j < n && s[i - j] == s[i + j]) j++;
6 res[i] = j;
7 k = 1;
8 while (i - k >= 0 && i + k < n && k + res[i - k] < j) {
9 res[i + k] = res[i - k];
10 k++;
11 }
12 i += k;
13 j -= k;
14 }
15 return res;
16}
各位置から始まる接尾辞と文字列全体の最長共通接頭辞長
s = ababa -> {5, 0, 3, 0, 1}
res[0] = N（空文字列なら空配列）
計算量 O(N)
17vector<int> z_algorithm(const string& s) {
18 int n = s.size(), i = 1, j = 0, k;
19 vector<int> res(n);
20 while (i < n) {
21 while (i + j < n && s[i + j] == s[j]) j++;
22 res[i] = j;
23 for (k = 1; i + k < n && k + res[k] < j; k++) res[i + k] = res[k];
24 i += k;
25 j -= min(j, k);
26 }
27 if (n) res[0] = n;
28 return res;
29}
