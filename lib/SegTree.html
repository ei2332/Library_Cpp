SegTree

モノイドによる区間集約と点更新を行うセグメント木。lazyスコープでは区間更新にも対応する。

1template <class T = int>
2struct SegTree {
3T f1(T x, T y) { return x + y; }
4T e = 0;
5using S = pair<T, T>;
6T f2(T x, S f, int len) { return f.first * x + f.second * len; }
7S f3(S o, S n) { return {o.first * n.first, n.first * o.second + n.second}; }
8S le = {1, 0};
9
10vector<T> d;
11vector<S> lz;
12int sz = 1, log = 0;
13SegTree(int n) {
14while (sz < n) sz *= 2, log++;
15d.assign(2 * sz, e);
16lz.assign(sz, le);
17}
18void push(int k, int len) {
19if (lz[k] == le) return;
20int l = k * 2, r = l + 1;
21d[l] = f2(d[l], lz[k], len / 2);
22d[r] = f2(d[r], lz[k], len / 2);
23if (l < sz) {
24lz[l] = f3(lz[l], lz[k]);
25lz[r] = f3(lz[r], lz[k]);
26}
27lz[k] = le;
28}
29void set(int i, T x) {
30int k = i + sz;
31for (int h = log; h > 0; h--) push(k >> h, 1LL << h);
32d[k] = x;
33while (k >>= 1) d[k] = f1(d[k * 2], d[k * 2 + 1]);
34}
35T query(int l, int r, int k = 1, int a = 0, int b = -1) {
36if (b < 0) b = sz;
37if (r <= a || b <= l) return e;
38if (l <= a && b <= r) return d[k];
39push(k, b - a);
40int m = (a + b) / 2;
41return f1(query(l, r, k * 2, a, m), query(l, r, k * 2 + 1, m, b));
42}
43void apply(int l, int r, S f, int k = 1, int a = 0, int b = -1) {
44if (b < 0) b = sz;
45if (r <= a || b <= l) return;
46if (l <= a && b <= r) {
47d[k] = f2(d[k], f, b - a);
48if (k < sz) lz[k] = f3(lz[k], f);
49return;
50}
51push(k, b - a);
52int m = (a + b) / 2;
53apply(l, r, f, k * 2, a, m);
54apply(l, r, f, k * 2 + 1, m, b);
55d[k] = f1(d[k * 2], d[k * 2 + 1]);
56}
57T get(int i) { return query(i, i + 1); }
58};

説明

3T f1(T x, T y) { return x + y; }

f1: モノイドの二項演算

4T e = 0;

e: f1の単位元

6T f2(T x, S f, int len) { return f.first * x + f.second * len; }

f2: 区間集約値に作用を適用する関数

f2(f1(x,y),f,n+m) = f1(f2(x,f,n),f2(y,f,m))

f2(x,le,len) = x

7S f3(S o, S n) { return {o.first * n.first, n.first * o.second + n.second}; }

f3: モノイドの二項演算（古い作用oの後に新しい作用nを適用）

f2(f2(x,o,len),n,len) = f2(x,f3(o,n),len)

8S le = {1, 0};

le: f3の単位元

13SegTree(int n) {

計算量 O(n)

18void push(int k, int len) {

ノードkに蓄積された作用を左右の子へ伝搬する

計算量 O(1)

29void set(int i, T x) {

i番目の要素をxに変更する

計算量 O(log n)

35T query(int l, int r, int k = 1, int a = 0, int b = -1) {

半開区間[l,r)の集約値を返す

計算量 O(log n)

43void apply(int l, int r, S f, int k = 1, int a = 0, int b = -1) {

半開区間[l,r)に作用fを適用する

計算量 O(log n)

i番目の要素を返す
