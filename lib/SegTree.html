SegTree

モノイドによる区間集約と点更新を行うセグメント木。lazyスコープでは区間更新にも対応する。

1template <class T = int>
2struct SegTree {
f1: モノイドの二項演算
3 T f1(T x, T y) { return x + y; }
e: f1の単位元
4 T e = 0;
5 using S = pair<T, T>;
f2: 区間集約値に作用を適用する関数
f2(f1(x,y),f,n+m) = f1(f2(x,f,n),f2(y,f,m))
f2(x,le,len) = x
6 T f2(T x, S f, int len) { return f.first * x + f.second * len; }
f3: モノイドの二項演算（古い作用oの後に新しい作用nを適用）
f2(f2(x,o,len),n,len) = f2(x,f3(o,n),len)
7 S f3(S o, S n) { return {o.first * n.first, n.first * o.second + n.second}; }
le: f3の単位元
8 S le = {1, 0};
9
10 vector<T> d;
11 vector<S> lz;
12 int sz = 1, log = 0;
計算量 O(n)
13 SegTree(int n) {
14 while (sz < n) sz *= 2, log++;
15 d.assign(2 * sz, e);
16 lz.assign(sz, le);
17 }
ノードkに蓄積された作用を左右の子へ伝搬する
計算量 O(1)
18 void push(int k, int len) {
19 if (lz[k] == le) return;
20 int l = k * 2, r = l + 1;
21 d[l] = f2(d[l], lz[k], len / 2);
22 d[r] = f2(d[r], lz[k], len / 2);
23 if (l < sz) {
24 lz[l] = f3(lz[l], lz[k]);
25 lz[r] = f3(lz[r], lz[k]);
26 }
27 lz[k] = le;
28 }
i番目の要素をxに変更する
計算量 O(log n)
29 void set(int i, T x) {
30 int k = i + sz;
31 for (int h = log; h > 0; h--) push(k >> h, 1LL << h);
32 d[k] = x;
33 while (k >>= 1) d[k] = f1(d[k * 2], d[k * 2 + 1]);
34 }
半開区間[l,r)の集約値を返す
計算量 O(log n)
35 T query(int l, int r, int k = 1, int a = 0, int b = -1) {
36 if (b < 0) b = sz;
37 if (r <= a || b <= l) return e;
38 if (l <= a && b <= r) return d[k];
39 push(k, b - a);
40 int m = (a + b) / 2;
41 return f1(query(l, r, k * 2, a, m), query(l, r, k * 2 + 1, m, b));
42 }
半開区間[l,r)に作用fを適用する
計算量 O(log n)
43 void apply(int l, int r, S f, int k = 1, int a = 0, int b = -1) {
44 if (b < 0) b = sz;
45 if (r <= a || b <= l) return;
46 if (l <= a && b <= r) {
47 d[k] = f2(d[k], f, b - a);
48 if (k < sz) lz[k] = f3(lz[k], f);
49 return;
50 }
51 push(k, b - a);
52 int m = (a + b) / 2;
53 apply(l, r, f, k * 2, a, m);
54 apply(l, r, f, k * 2 + 1, m, b);
55 d[k] = f1(d[k * 2], d[k * 2 + 1]);
56 }
i番目の要素を返す
計算量 O(log n)
57 T get(int i) { return query(i, i + 1); }
58};
