2bool isprime(ll n) {
3if (n < 3) return n == 2;
4if (!(n & 1)) return false;
5
6ll d = n - 1;
7int s = 0;
8while (!(d & 1)) d >>= 1, s++;
9
10for (ll a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
11a %= n;
12if (!a) continue;
13
14ll x = 1, v = a, e = d;
15while (e) {
16if (e & 1) x = (__int128)x * v % n;
17v = (__int128)v * v % n;
18e >>= 1;
19}
20
21if (x == 1) continue;
22for (int i = 0; i < s; i++) {
23if (x == n - 1) break;
24x = (__int128)x * x % n;
25}
26if (x != n - 1) return false;
27}
28return true;

32SPF(int n) : t(n + 1) {
33for (int i = 2; i <= n; i++) {
34if (!t[i]) {
35p.push_back(i);
36t[i] = i;
37}
38for (auto& j : p) {
39if (j > t[i] || i > n / j) break;
40t[i * j] = j;
41}
42}
43}

48ll rho(ll n) {
49if (1 - n % 2) return 2;
50for (ll c = 1;; c++) {
51auto f = [&](ll x) -> ll { return ((__int128)x * x + c) % n; };
52ll x = 2, y = 2;
53while (true) {
54x = f(x), y = f(f(y));
55ll d = gcd(abs(x - y), n);
56if (d == n) break;
57if (d - 1) return d;
58}
59}
60}
