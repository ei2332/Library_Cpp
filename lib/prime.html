1using ll = long long;
符号付き64bit整数の素数判定
計算量 O(log n)
2bool isprime(ll n) {
3 if (n < 3) return n == 2;
4 if (!(n & 1)) return false;
5
6 ll d = n - 1;
7 int s = 0;
8 while (!(d & 1)) d >>= 1, s++;
9
10 for (ll a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
11 a %= n;
12 if (!a) continue;
13
14 ll x = 1, v = a, e = d;
15 while (e) {
16 if (e & 1) x = (__int128)x * v % n;
17 v = (__int128)v * v % n;
18 e >>= 1;
19 }
20
21 if (x == 1) continue;
22 for (int i = 0; i < s; i++) {
23 if (x == n - 1) break;
24 x = (__int128)x * x % n;
25 }
26 if (x != n - 1) return false;
27 }
28 return true;
29}
最小素因数表と素数列をO(N)で構築する
計算量 O(N)
30struct SPF {
31 vector<int> p, t;
32 SPF(int n) : t(n + 1) {
33 for (int i = 2; i <= n; i++) {
34 if (!t[i]) {
35 p.push_back(i);
36 t[i] = i;
37 }
38 for (auto& j : p) {
39 if (j > t[i] || i > n / j) break;
40 t[i * j] = j;
41 }
42 }
43 }

合成数nの非自明な約数を1つ返す
48ll rho(ll n) {
49 if (1 - n % 2) return 2;
50 for (ll c = 1;; c++) {
51 auto f = [&](ll x) -> ll { return ((__int128)x * x + c) % n; };
52 ll x = 2, y = 2;
53 while (true) {
54 x = f(x), y = f(f(y));
55 ll d = gcd(abs(x - y), n);
56 if (d == n) break;
57 if (d - 1) return d;
58 }
59 }
60}
