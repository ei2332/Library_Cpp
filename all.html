15bool bfs(int s, int t) {
16fill(l.begin(), l.end(), -1);
17queue<int> q;
18l[s] = 0;
19q.push(s);
20while (!q.empty()) {
21int v = q.front();
22q.pop();
23for (auto& i : e[v]) {
24if (i.cap > 0 && l[i.to] == -1) {
25l[i.to] = l[v] + 1;
26q.push(i.to);
27}
28}
29}
30return l[t] != -1;
31}
32int dfs(int v, int t, int f) {
33if (v == t) return f;
34for (int& i = it[v]; i < (int)e[v].size(); i++) {
35auto& x = e[v][i];
36if (x.cap == 0 || l[x.to] != l[v] + 1) continue;
37int d = dfs(x.to, t, min(f, x.cap));
38if (d) {
39x.cap -= d;
40e[x.to][x.rev].cap += d;
41return d;
42}
43}
44return 0;
45}
46int max_flow(int s, int t) {
47int ans = 0;
48while (bfs(s, t)) {
49fill(it.begin(), it.end(), 0);
50while (int f = dfs(s, t, INT_MAX)) ans += f;
51}
52return ans;
53}
54};
